### 🔌 WordPress Custom Development

```php
// Example: Custom REST API Endpoint with CMB2 Integration
add_action('rest_api_init', function() {
    register_rest_route('custom/v1', '/projects', [
        'methods' => 'GET',
        'callback' => 'get_custom_projects',
        'permission_callback' => '__return_true'
    ]);
});

function get_custom_projects() {
    $args = [
        'post_type' => 'project',
        'posts_per_page' => 10,
        'meta_query' => [
            [
                'key' => '_project_featured',
                'value' => '1',
                'compare' => '='
            ]
        ]
    ];
    
    $projects = get_posts($args);
    $response = [];
    
    foreach($projects as $project) {
        $response[] = [
            'id' => $project->ID,
            'title' => get_the_title($project->ID),
            'metadata' => get_post_meta($project->ID, '_project_details', true)
        ];
    }
    
    return rest_ensure_response($response);
}
```
